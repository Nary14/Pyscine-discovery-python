Module 4 — ex05 · Invert case of each character

#!/usr/bin/env python3

word = input("Enter a word! :")
res = ""
for i in word:
    if i.isupper():
        res += i.lower()
    elif i.islower():
        res += i.upper()
    else:
        res += i
print(res)

# Input:  "Hello42"
# Output: hELLO42

Module 5 — ex03 · Set comprehension

#!/usr/bin/env python3

Original_array = [2, 8, 9, 48, 8, 22, -12, 2]
print("Original array: ", Original_array)

new_array = {x + 2 for x in Original_array if x > 5}
print("New array: ", new_array)

# Original array: [2, 8, 9, 48, 8, 22, -12, 2]
# New array: {10, 11, 50, 24}
